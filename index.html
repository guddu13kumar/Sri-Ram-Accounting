function calcBill(isTypingPaid = false) { let totalTaxable = 0; let totalGstAmt = 0; let rawGrandTotal = 0; let totalPurchaseCost = 0; const isInclusive = document.querySelector('input[name="tax_calc_type"]:checked')?.value === 'inclusive'; document.querySelectorAll('#bill-items-body tr').forEach(row => { const name = row.querySelector('.b-name').value.trim().toUpperCase(); const rate = Number(row.querySelector('.b-rate').value) || 0; const qty = Number(row.querySelector('.b-qty').value) || 0; const gst = Number(row.querySelector('.b-gst').value) || 0; let grossLine = rate * qty; let lineTaxable = 0; let lineGst = 0; let lineTotal = 0; if (isInclusive && gst > 0) { lineTaxable = grossLine / (1 + (gst / 100)); lineGst = grossLine - lineTaxable; lineTotal = grossLine; } else { lineTaxable = grossLine; lineGst = lineTaxable * (gst / 100); lineTotal = lineTaxable + lineGst; } if(currentMode === "SALE" && name) { const prod = productsDB.find(p => p.name === name); if(prod) { totalPurchaseCost += (prod.pur * qty); if (prod.stock < qty) row.querySelector('.b-qty').classList.add('stock-warning'); else row.querySelector('.b-qty').classList.remove('stock-warning'); } } row.querySelector('.b-amt').innerText = (Math.round(lineTotal * 100) / 100).toLocaleString('en-IN', {minimumFractionDigits: 2}); totalTaxable += lineTaxable; totalGstAmt += lineGst; rawGrandTotal += lineTotal; }); const discount = Number(document.getElementById('bill-discount').value) || 0; rawGrandTotal = Math.max(0, rawGrandTotal - discount); const roundGrandTotal = Math.round(rawGrandTotal); const profit = currentMode === "SALE" ? (roundGrandTotal - totalGstAmt - totalPurchaseCost) : 0; document.getElementById('bill-taxable').innerText = "₹ " + (Math.round(totalTaxable * 100) / 100).toLocaleString('en-IN', {minimumFractionDigits: 2}); document.getElementById('bill-gst').innerText = "₹ " + (Math.round(totalGstAmt * 100) / 100).toLocaleString('en-IN', {minimumFractionDigits: 2}); document.getElementById('bill-grand-total').innerText = "₹ " + roundGrandTotal.toLocaleString('en-IN'); const payInput = document.getElementById('bill-paid'); let rawPaid = 0; if(!isTypingPaid) { payInput.value = roundGrandTotal; rawPaid = roundGrandTotal; } else { rawPaid = Number(payInput.value) || 0; } const due = Math.max(0, roundGrandTotal - rawPaid); document.getElementById('bill-due').innerText = "₹ " + due.toLocaleString('en-IN'); return { totalTaxable, totalGstAmt, grandTotal: roundGrandTotal, rawPaid, due, profit, discount }; const rate = Number(row.querySelector('.b-rate').value) || 0; const qty = Number(row.querySelector('.b-qty').value) || 1; const gst = Number(row.querySelector('.b-gst').value) || 0; const imeiMain = row.querySelector('.b-imei-main').value.trim().toUpperCase(); let validImeis = []; if(imeiMain) validImeis.push(imeiMain); row.querySelectorAll('.b-imei-extra').forEach(inp => { if(inp.value.trim()) validImeis.push(inp.value.trim().toUpperCase()); }); const imeiStr = validImeis.join(', '); if(name && rate > 0) { let prod = productsDB.find(p => p.name === name); if (currentMode === "SALE" && prod) { if (prod.stock < qty) { errorMsg = `Not enough stock for ${name}. Available: ${prod.stock}`; break; } if (prod.track) { if (validImeis.length !== qty) { errorMsg = `Please enter exactly ${qty} 15-digit IMEI(s) for ${name}`; break; } let missing = validImeis.find(im => !prod.imeiList.includes(im)); if(missing) { errorMsg = `IMEI ${missing} not found in stock for ${name}!`; break; } } } const isInclusive = document.querySelector('input[name="tax_calc_type"]:checked')?.value === 'inclusive'; let grossLine = rate * qty; let lineTaxable = 0; let lineGst = 0; let lineTotal = 0; if (isInclusive && gst > 0) { lineTaxable = grossLine / (1 + (gst / 100)); lineGst = grossLine - lineTaxable; lineTotal = grossLine; } else { lineTaxable = grossLine; lineGst = lineTaxable * (gst / 100); lineTotal = lineTaxable + lineGst; } tempItems.push({name, rate, qty, gst, imei: imeiStr, amt: Math.round(lineTotal)}); if(prod) tempDeductions.push({ prod, qty, validImeis }); } else if (name) { errorMsg = `Invalid rate for ${name}`; break; } } if(errorMsg) return showToast("Stock Error", errorMsg, "error"); if(tempItems.length === 0) return showToast("Item Error", "Add at least one valid product!", "error"); isSaving = true; const btnSave = document.getElementById('btn-save-bill'); if(btnSave) { btnSave.innerHTML = "⏳ Saving..."; btnSave.classList.add("opacity-50", "cursor-not-allowed"); } tempDeductions.forEach(d => { if(currentMode === "SALE") { d.prod.stock -= d.qty; if(d.prod.track && d.prod.imeiList) { d.prod.imeiList = d.prod.imeiList.filter(i => !d.validImeis.includes(i)); } } if(currentMode === "PURCHASE") { d.prod.stock += d.qty; if(d.prod.track) { if(!d.prod.imeiList) d.prod.imeiList = []; d.prod.imeiList.push(...d.validImeis); } } }); // Smart Ledger Logic let activePaymentMode = "CASH"; if(totals.rawPaid === 0) activePaymentMode = "CREDIT"; else if (totals.rawPaid < totals.grandTotal) activePaymentMode = "PARTIAL"; const billRecord = { id: 'INV-'+Date.now(), invNo: invNo, date: bDate, type: currentMode, party: party, mobile: mobile, items: tempItems, basic: totals.totalTaxable, gst: totals.totalGstAmt, discount: totals.discount, total: totals.grandTotal, paid: totals.rawPaid, due: totals.due, paymentMode: activePaymentMode, finance: null, profit: totals.profit }; if(pData) { if(currentMode === "SALE") pData.ob += totals.due; if(currentMode === "PURCHASE") pData.ob -= totals.due; Storage.set('sre_parties', partiesDB); } } showPopup("Saved", currentMode + " Bill Saved Successfully!"); generateInvoiceHTML(billRecord); setTimeout(() => { isSaving = false; if(btnSave) { btnSave.innerHTML = `(Ctrl+S) 💾 Save`; btnSave.classList.remove("opacity-50", "cursor-not-allowed"); } }, 2000); }
